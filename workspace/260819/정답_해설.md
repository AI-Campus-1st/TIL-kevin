## 1. 뒤죽박죽인 날짜 표기를 하나로 통일하기

### 정답
```py
import re
import datetime

# 순서가 중요
PATTERNS = [
    # 혹시 모를 공백 고려 사이사이에 \s* 적용

    # 2024년 12월 24일 / 2024년 3월 5일
    ("ymd_kor",   re.compile(r"(?P<y>\d{4})\s*년\s*(?P<m>\d{1,2})\s*월\s*(?P<d>\d{1,2})\s*일")),

    # 2024.12.24 / 2024-12-24 / 2024/12/24
    ("ymd_sep",   re.compile(r"(?P<y>\d{4})[.\-/](?P<m>\d{1,2})[.\-/](?P<d>\d{1,2})")),

    # 12/24/2024  (미국식 — 4자리 연도가 뒤에 온다)
    ("mdy_slash", re.compile(r"(?P<m>\d{1,2})/(?P<d>\d{1,2})/(?P<y>\d{4})")),

    # 24.12.24  (두 자리 연도)
    ("ymd_short", re.compile(r"(?<!\d)(?P<y>\d{2})[.\-/](?P<m>\d{1,2})[.\-/](?P<d>\d{1,2})(?!\d)")),
]

def normalize_date(s):
    if not s or not str(s).strip():
        return (None, "빈 값")

    for tag, pat in PATTERNS:
        m = pat.search(str(s))
        if not m:
            continue

        y, mo, d = int(m["y"]), int(m["m"]), int(m["d"])
        if tag == "ymd_short":
            y += 2000 if y < 70 else 1900        # 두 자리 연도 해석 규칙

        # 조건 5 — 실제 존재하는 날짜인지 (후처리)
        try:
            dt = datetime.date(y, mo, d).strftime('%Y-%m-%d')
        except ValueError:
            return (None, f"{tag} · 유효하지 않은 날짜")

        return (dt, tag)

    return (None, "매치 없음")

samples = [
    "2024.12.24", "2024-12-24", "2024/12/24",
    "24.12.24", "2024년 12월 24일", "2024년 3월 5일",
    "12/24/2024", "2024.12.24 14:30", "등록일 : 2024.12.24",
    "2024-13-45", "작성일 없음", "",
]
for s in samples:
    value, tag = normalize_date(s)
    print(f"{s:22s} → {str(value):12s} [{tag}]")
```

### 해설

- **명명 그룹의 가독성** `m.group(1)`, `m.group(3)`으로 코드를 작성하면 나중에 확인했을 때 해당 순서가 무엇을 의미하는지 알기 어렵습니다. `(?P<y>\d{4})`로 이름을 붙이면 매치 객체에서 키로 활용할 수 있습니다.

- `2024-13-45`는 `\d{4}-\d{1,2}-\d{1,2}` 패턴에 일치하지만, 13월 45일은 실제로 불가능한 날짜로, **정규표현식으로 확인할 수 없습니다.** 따라서, 후처리로 실제로 존재하는 날짜인지 datetime 모듈을 활용하여 확인합니다. try-except문에서 실패하면 존재하지 않는 날짜입니다.

- 6번째 조건을 위해 **앞 숫자가 4자리인지, 뒤에 있는 숫자가 4자리인지** 패턴을 구분해야합니다. 단순하게 `\d+/\d+/\d+`로 처리하면 잘못 처리될 위험이 있습니다.

- 두 자리 연도 패턴의 경우 후방 탐색을 설정하여 숫자가 앞뒤에 오지 않도록 처리해야 합니다.  두 자리 연도 패턴을 그냥 사용하게 되면 `2024.12.24`의 뒷부분 `24.12.24`에도 매치가 되기 때문에 위험할 수 있습니다. 패턴 순서상 앞에서 이미 확인이 되지만, 정확한 사용이 중요합니다.


## 2. 서버 액세스 로그 파싱과 집계

### 정답
```py
import re
import pandas as pd
from collections import Counter

LOG = re.compile(
    r'^(?P<ip>\d{1,3}(?:\.\d{1,3}){3})'           # IP — 비캡처 그룹으로 반복
    r'\s+\S+\s+\S+\s+'                            # ident, authuser (보통 - -) 공백과 공백이 아닌것 순서로 매칭
    r'\[(?P<timestamp>[^\]]+)\]\s+'               # [일시] 부분
    r'"(?P<method>[A-Z]+)\s+(?P<path>\S+)[^"]*"\s+'
    r'(?P<status>\d{3})\s+'
    r'(?P<bytes>\d+)\s+'                          
    r'"[^"]*"\s+'                                 # referer (쓰지 않음)
    r'"(?P<user_agent>[^"]*)"'
)

BOT = re.compile("bot|crawler|spider|python-requests", re.IGNORECASE)

def parse_log(path: str) -> tuple[pd.DataFrame, int]:
    rows, skipped = [], 0
    with open(path, encoding="utf-8") as f:
        for line in f.readlines():
            line = line.strip()
            if not line:
                continue
            m = LOG.match(line)
            if not m:
                skipped += 1
                continue
            # 매치된 변수: 값 형태를 dict로 받음
            rows.append(m.groupdict())

    df = pd.DataFrame(rows)
    if len(df):
        df["status"] = df["status"].astype(int)
        df["bytes"] = df["bytes"].astype(int)
        # 대소문자를 구분없이 특정 단어 포함여부 체크하는 코드
        df["is_bot"] = df["user_agent"].str.contains(BOT, na=False)
        # 같은 역할
        df["is_bot"] = df["user_agent"].map(
            lambda x: any(
                b in x.lower() for b in "bot|crawler|spider|python-requests".split('|')
            )
        )
    return df, skipped

df, skipped = parse_log("access.log")
print(f"파싱 {len(df)}줄 · 건너뜀 {skipped}줄")

print("\n── 상태코드별 요청 수 ──")
print(df["status"].value_counts().sort_index().to_string())

print("\n── 4xx·5xx 발생 경로 상위 5 ──")
bad = df[df["status"] >= 400]
print(bad["path"].value_counts().head(5).to_string() or "  없음")

print("\n── 봇 의심 User-Agent ──")
bots = df[df["is_bot"]]["user_agent"].value_counts()
print(bots.to_string() if not bots.empty else "  없음")
print(f"  봇 요청 비율 {df['is_bot'].mean() * 100:.1f}%")

df.to_csv("access_report.csv", index=False, encoding="utf-8-sig")
```

### 해설
- 패턴을 작성하는 것은 어려울 수 있지만, 로그처럼 형식이 고정된 텍스트는 **한 패턴**으로 한 번에 잡는 것이 정확하고 빠릅니다.

- 명명 그룹에서 값을 `m.groupdict()`로 dict를 뽑을 수 있습니다. 이를 모아서 `DataFrame`에 넣으면 쉽게 데이터프레임을 만들 수 있습니다.

- 맨 처음 ip 주소를 탐색하는 패턴은 **비캡처 그룹이라고 합니다.** IP를 `\d{1,3}(\.\d{1,3}){3}`로 쓰면 그룹이 하나 더 생기고 `groupdict()`에는 영향이 없지만, `findall()`을 사용하면 결과가 달라집니다.
```py
# LOG.findall(line) 호출했을 때,

# \d{1,3}(\.\d{1,3}){3}
('203.0.113.42', '.42', ...)
# \d{1,3}(?:\.\d{1,3}){3}
('203.0.113.42', '06/Aug/2026:14:22:31', ...)
```
