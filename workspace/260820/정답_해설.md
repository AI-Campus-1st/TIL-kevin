## 1. 네이버 연관검색어 수집 함수 만들기

### 정답
```py
import requests
import json

SEARCH = "https://ac.search.naver.com/nx/ac"
HEADERS = {}

def get_related_keywords(keyword: str) -> list[str]:
    """키워드의 연관검색어를 리스트로 돌려준다. 없으면 빈 리스트."""
    try:
        r = requests.get(SEARCH, params={"q": keyword, "_callback": "_jsonp_7", 'q_enc': 'UTF-8', 'st': '100'},
                         headers=HEADERS, timeout=10)
        r.raise_for_status()
    except requests.RequestException as e:
        print(f"요청 실패: {e}")
        return []
    
    # 응답은 json처럼 보이지만, 앞 뒤에 있는 _jsonp_7(, )로 인해 제대로 파싱이 되지 않는 문제가 있음
    result = json.loads(r.text.split('(', maxsplit=1)[-1][:-1])
    return [item[0].strip() for item in result['items'][0]]

if __name__ == "__main__":
    for kw in ["부트캠프", "웹크롤링", "ㅁㄴㅇㄹ"]:
        result = get_related_keywords(kw)
        print(f"{kw:8s} → {len(result)}건 {result[:5]}")
```

### 해설

1. F12 개발자도구의 `Network` 탭을 열고 네이버 검색어 입력창에 값을 입력하면 문서가 쌓입니다.
2. 쌓이는 문서 중 하나를 선택해 preview를 확인하면 연관검색어들이 들어가 있고, 이 URL을 가져와 재현합니다.

- 다만, 해당 문서에는 _jsonp_7()로 감싸져있어, `r.json()`으로 바로 파이썬 딕셔너리로 파싱할 수 없기 때문에, `r.text`로 문자열을 가져온 뒤 슬라이싱하여 파싱이 가능한 문자열로 바꾸어야 하는게 핵심입니다. 이 때 파싱은 `json.loads(text)`로 합니다.

## 2. 네이버 웹툰 전체 목록 수집

### 정답
```py
import requests
import pandas as pd

API = "https://comic.naver.com/api/webtoon/titlelist/weekday"
DETAIL = "https://comic.naver.com/webtoon/list?titleId="

HEADERS = {
    "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64)",
}

WEEKDAY_KR = {
    "MONDAY": "월", "TUESDAY": "화", "WEDNESDAY": "수", "THURSDAY": "목",
    "FRIDAY": "금", "SATURDAY": "토", "SUNDAY": "일", "DAILY_PLUS": "매일+",
}

def fetch_all() -> list[dict]:
    r = requests.get(API, headers=HEADERS, timeout=10)
    r.raise_for_status()
    data = r.json()

    rows = []
    # 응답 구조는 Preview 탭에서 확인한 뒤 맞춰 쓴다
    for key, toons in data.get("titleListMap", {}).items():
        day = WEEKDAY_KR.get(key, key)
        for t in toons:
            rows.append({
                "제목": t.get("titleName", ""),
                "링크": DETAIL + str(t.get("titleId", "")),
                "요일": day,
            })
    return rows

rows = fetch_all()
df = pd.DataFrame(rows).drop_duplicates(subset=["링크", "요일"])
df.to_csv("naver_webtoon.csv", index=False, encoding="utf-8-sig")
print(f"{len(df)}건 수집 · 요일 분포\n{df['요일'].value_counts()}")
```

### 해설

- 이 과제는 **JSON 응답의 중첩 구조를 읽는 연습**입니다.
- 응답이 `{"titleListMap": {"MONDAY": [...], "TUESDAY": [...]}}` 처럼 요일이 키가 되고 값이 리스트인 형태라면, 바깥은 딕셔너리 순회, 안쪽은 리스트 순회로 이중 반복문이 됩니다.
- 응답에 있는 id만 가지고 상세 링크를 만드는 것은 대부분의 서비스에서 사용중인 방식입니다. id만 가지고 상세페이지 URL과 조립하면 상세페이지 URL을 쉽게 얻을 수 있습니다.

## 3. 사람인 채용공고 10페이지 수집

### 정답
```py
import time
import requests
import pandas as pd
from bs4 import BeautifulSoup

LIST_URL = "https://www.saramin.co.kr/zf_user/jobs/public/list"
HEADERS = {
    "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64)",
    "X-Requested-With": "XMLHttpRequest",
}
DELAY = 0.7
PAGES = 10

def text_of(node, selector, default=""):
    el = node.select_one(selector)
    return el.text.strip() if el else default

def fetch(page: int) -> dict:
    r = requests.get(LIST_URL, params={"page": page, "isAjaxRequest": "y"},
                     headers=HEADERS, timeout=10)
    r.raise_for_status()
    return r.json()

def parse(data: dict) -> list[dict]:
    """공고 한 건의 경계를 먼저 잡고, 그 안에서 재탐색한다."""
    html = data['innerHTML']
    soup = BeautifulSoup(html, "html.parser")
    rows = []
    for box in soup.select("div.list_body div.list_item"):
        keywords = [k.text.strip() for k in box.select("span.job_sector span")]
        rows.append({
            "기업명": text_of(box, "a.str_tit"),
            "그룹사": text_of(box, "span.main_corp"),
            "기업종류": text_of(box, "span.info_stock"),
            "공고명": text_of(box, "div.notification_info a.str_tit"),
            "직무키워드": keywords,
            "학력": text_of(box, "p.education"),
            "경력구분": text_of(box, "p.career"),
            "근무지": text_of(box, "p.work_place"),
        })
    return rows

result = []
for page in range(1, PAGES + 1):
    try:
        rows = parse(fetch(page))
    except Exception as e:
        print(f"{page}페이지 실패: {e}")
        continue

    if not rows:
        print(f"{page}페이지가 비어 있어 중단합니다.")
        break

    result.extend(rows)
    print(f"{page}페이지 · 누적 {len(result)}건")
    time.sleep(DELAY)

df = pd.DataFrame(result)
df["직무키워드"] = df["직무키워드"].map(lambda x: ", ".join(x))
df.to_csv("saramin.csv", index=False, encoding="utf-8-sig")
print(f"최종 {len(df)}건 저장")
```

### 해설

- 핵심은 json 포맷으로 먼저 파싱하고, HTML로 파싱하는 두 단계를 거쳐서 데이터를 추출하는 것입니다. 이 때 HTML 부분은 `Elements` 탭에서 분석하고 적용하면, 잘 나오는 것을 확인할 수 있습니다.

- **결측 방어** 그룹사가 없는 중소기업, 기업종류 표시가 없는 공고가 실제로 섞여 있습니다. `box.select_one(...).text`로 쓰면 해당 공고는 `AttributeError`를 발생시킵니다. `text_of()` 같은 헬퍼로 기본값을 주어 해결합니다.
