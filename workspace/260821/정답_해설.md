## 1. 네이버 VIBE Top 100 수집

### 정답
```py
import requests
import pandas as pd

API = "https://apis.naver.com/vibeWeb/musicapiweb/vibe/v1/chart/track/total"

HEADERS = {
    "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64)",

    # Accept를 주지 않으면 응답이 XML 형식으로 옴
    "Accept": "application/json",
}


def fetch_chart(start=1, display=100) -> list[dict]:
    r = requests.get(API, params={"start": start, "display": display},
                     headers=HEADERS, timeout=10)
    r.raise_for_status()
    data = r.json()

    # Preview 탭에서 확인한 경로를 그대로 옮긴다
    items = data["response"]["result"]["chart"]["items"]["tracks"]

    rows = []
    for i, t in enumerate(items, start=start):
        artists = [a.get("artistName", "") for a in t.get("artists", [])]
        rows.append({
            "순위": i,
            "곡명": t.get("trackTitle", ""),
            "아티스트": artists,
        })
    return rows

rows = fetch_chart()
assert len(rows) == 100, f"100건이 아닙니다: {len(rows)}건"

df = pd.DataFrame(rows)
df["아티스트"] = df["아티스트"].map(lambda x: ", ".join(x))
df.to_csv("vibe_top100.csv", index=False, encoding="utf-8-sig")
print(df.head())
```

### 해설

- `Header`에 `Accept`를 주지 않으면 응답이 XML로 옵니다. JSON으로 파싱하기 위해선 해당 옵션을 줘야합니다.

- **아티스트를 리스트**로 담는 데에는 이유가 있습니다. `Golden`처럼 협업곡은 아티스트가 두 명 이상이고, 응답에서도 배열로 오기 대문에 문자열 하나로 처리하지 않고 배열로 관리합니다. **수집 시점에는 구조를 보존하는게 중요합니다.**

## 2. 삼성전자 일별 시세 1년치 수집

### 정답
```py
import time
import requests
import pandas as pd
from bs4 import BeautifulSoup
from datetime import datetime, timedelta
import re

URL = "https://finance.naver.com/item/sise_day.naver"
CODE = "005930"
HEADERS = {
    "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64)",
}
DELAY = 0.7
MAX_PAGE = 40
TODAY = datetime.now()


def fetch(page: int) -> str:
    r = requests.get(URL, params={"code": CODE, "page": page},
                     headers=HEADERS, timeout=10)
    r.raise_for_status()
    return r.text

def get_num(t: str) -> int:
    m = re.search(r'\d+', t.replace(',', '').strip())
    return int(m.group()) if m else None

def parse(html: str) -> list[dict]:
    soup = BeautifulSoup(html, "html.parser")
    rows = []
    for tr in soup.select("table.type2 tr"):
        td = tr.select("td")
        # 구분선 행 스킵
        if len(td) < 7:
            continue

        date = td[0].text.strip()
        direction = td[2].select_one("em.bu_p").text.strip()
        amount = get_num(td[2].text.strip())

        rows.append({
            "날짜": date,
            "종가": get_num(td[1].text),
            "전일비": f"{direction} {amount}",
            "시가": get_num(td[3].text),
            "고가": get_num(td[4].text),
            "저가": get_num(td[5].text),
            "거래량": get_num(td[6].text),
        })
    return rows

result = []

for page in range(1, MAX_PAGE + 1):
    rows = parse(fetch(page))

    # 빈 응답인 경우 종료
    if not rows:
        print(f"{page}페이지가 비어 있습니다. 종료.")
        break

    # 마지막 날짜가 1년 이전일 경우 종료
    last_date = rows[0]['날짜'] # 가장 최신 날짜가 맨 앞에 있음
    if datetime.strptime(last_date, '%Y.%m.%d') < TODAY - timedelta(days=365):
        print(f"{last_date}, 1년 이전 날짜에 도달했습니다. 종료.")
        break

    result.extend(rows)
    print(f"{page}페이지 · 누적 {len(result)}건 · 최종 {rows[-1]['날짜']}")
    time.sleep(DELAY)

df = pd.DataFrame(result).drop_duplicates(subset=["날짜"])
df.to_csv("samsung_1y.csv", index=False, encoding="utf-8-sig")
print(f"\n최종 {len(df)}건 · {df['날짜'].min()} ~ {df['날짜'].max()}")
```

### 해설

- 숫자를 정제하여 `,`와 같은 불필요한 기호를 제거하고, 나중에 활용하기 좋도록 변환합니다. 이 때, `get_num()` 헬퍼 함수를 활용하여 가격 필드들에 대해 적용할 수 있습니다.

- 종료 조건은 `datetime` 모듈을 활용하여 크롤링한 날짜는 `datetime` 객체로 변환, 현재 날짜와 `timedelta(days=365)`를 활용해 1년 이전인지 여부를 판단합니다.

- 전일비의 경우에는 화면상에선 보이지 않지만, `html` 상으로는 `em` 태그에 상승과 하락이라는 텍스트로 들어가 있어 이를 꺼내서 전일비에 붙여주면 됩니다.

## 3. 네이버 뉴스 검색기 함수 만들기 (스크롤 포함)

### 정답
```py
import time
import requests
import pandas as pd
from bs4 import BeautifulSoup

SEARCH = "https://s.search.naver.com/p/newssearch/3/api/tab/more"
HEADERS = {
    "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64)",
    "Referer": "https://search.naver.com/",
}
DELAY = 0.7
PER_PAGE = 10

def _parse(html: str) -> list[dict]:
    soup = BeautifulSoup(html, "html.parser")
    rows = []
    for item in soup.select("div.fds-news-item-list-tab > div"):
        title = item.select_one("a[data-heatmap-target=\".tit\"]")
        press = item.select_one("span.sds-comps-profile-info-title-text span")
        desc = item.select_one("span.sds-comps-text-type-body1")
        if title is None:
            continue
        rows.append({
            "제목": title.get("title") or title.text,
            "언론사": press.text if press else "",
            "링크": title.get("href", ""),
            "요약": desc.text if desc else "",
        })
    return rows

def crawl_naver_news(keyword: str, page: int = 1) -> list[dict]:
    """page=1 이면 첫 화면만, page=n 이면 n번째 묶음까지 누적 수집."""
    result, seen = [], set()

    for p in range(1, page + 1):
        # 스크롤이 바꾸는 값
        start = (p - 1) * PER_PAGE + 1
        params = {'query': keyword, 'ssc': 'tab.news.all', 'start': start, 
                  'sort': 1, # 0: 관련도순, 1: 최신순 
                  'ds': '2026.08.12', 'de': '2026.08.12'} 

        try:
            r = requests.get(SEARCH, params=params, headers=HEADERS, timeout=10)
            r.raise_for_status()
        except requests.RequestException as e:
            print(f"{p}번째 요청 실패: {e}")
            break

        rows = _parse(r.json()['collection'][0]['html'])
        if not rows:
            print(f"{p}번째에서 결과가 없어 종료합니다.")
            break

        new = [x for x in rows if x["링크"] not in seen]
        if not new:                            # 같은 결과가 반복되면 종료
            print(f"{p}번째가 이전과 동일합니다. 종료.")
            break

        seen |= {x["링크"] for x in new}
        result.extend(new)
        print(f"{p}번째 · 누적 {len(result)}건")
        time.sleep(DELAY)

    return result

keyword = "인공지능"
rows = crawl_naver_news(keyword, page=5)
pd.DataFrame(rows).to_csv(f"news_{keyword}.csv",
                            index=False, encoding="utf-8-sig")
print(f"{len(rows)}건 저장")
```

### 해설

- 본 과제는 어떤 parameter를 전달하여야 하는지 확인함에 있어서의 어려움과 복잡한 html에서 `selector`를 어떤 것을 사용해야되는지 찾는 것이 핵심입니다.

- 뉴스 하나를 담는 `div.fds-news-item-list-tab > div` 박스 태그를 찾고, 그 내부에서 `a[data-heatmap-target=\".tit\"]`로 표현되는 링크를 포함하는 제목 태그를 가져오는 셀렉터와, `span.sds-comps-profile-info-title-text span` 언론사 태그 셀렉터, `span.sds-comps-text-type-body1` 본문 태그 셀렉터를 찾으면 됩니다.
